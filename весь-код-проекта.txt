import { createStart, createCsrfMiddleware, createMiddleware } from "@tanstack/react-start";

import { renderErrorPage } from "./lib/error-page";
import { attachSupabaseAuth } from "./lib/supabase-auth-attacher";

const errorMiddleware = createMiddleware().server(async ({ next }) => {
  try {
    return await next();
  } catch (error) {
    if (error != null && typeof error === "object" && "statusCode" in error) {
      throw error;
    }
    console.error(error);
    return new Response(renderErrorPage(), {
      status: 500,
      headers: { "content-type": "text/html; charset=utf-8" },
    });
  }
});

// Start installs this automatically when src/start.ts is absent; defining the
// file opts out, so re-add it explicitly to keep server functions protected
// from cross-site requests.
const csrfMiddleware = createCsrfMiddleware({
  filter: (ctx) => ctx.handlerType === "serverFn",
});

export const startInstance = createStart(() => ({
  requestMiddleware: [errorMiddleware, csrfMiddleware],
  functionMiddleware: [attachSupabaseAuth],
}));

import "./lib/error-capture";

import { consumeLastCapturedError } from "./lib/error-capture";
import { renderErrorPage } from "./lib/error-page";

type ServerEntry = {
  fetch: (request: Request, env: unknown, ctx: unknown) => Promise<Response> | Response;
};

let serverEntryPromise: Promise<ServerEntry> | undefined;

async function getServerEntry(): Promise<ServerEntry> {
  if (!serverEntryPromise) {
    serverEntryPromise = import("@tanstack/react-start/server-entry").then(
      (m) => (m.default ?? m) as ServerEntry,
    );
  }
  return serverEntryPromise;
}

// h3 swallows in-handler throws into a normal 500 Response with body
// {"unhandled":true,"message":"HTTPError"} — try/catch alone never fires for those.
async function normalizeCatastrophicSsrResponse(response: Response): Promise<Response> {
  if (response.status < 500) return response;
  const contentType = response.headers.get("content-type") ?? "";
  if (!contentType.includes("application/json")) return response;

  const body = await response.clone().text();
  if (!isH3SwallowedErrorBody(body)) return response;

  console.error(consumeLastCapturedError() ?? new Error(`h3 swallowed SSR error: ${body}`));
  return new Response(renderErrorPage(), {
    status: 500,
    headers: { "content-type": "text/html; charset=utf-8" },
  });
}

function isH3SwallowedErrorBody(body: string): boolean {
  try {
    const payload = JSON.parse(body) as { unhandled?: unknown; message?: unknown };
    return payload.unhandled === true && payload.message === "HTTPError";
  } catch {
    return false;
  }
}

export default {
  async fetch(request: Request, env: unknown, ctx: unknown) {
    try {
      const handler = await getServerEntry();
      const response = await handler.fetch(request, env, ctx);
      return await normalizeCatastrophicSsrResponse(response);
    } catch (error) {
      console.error(error);
      return new Response(renderErrorPage(), {
        status: 500,
        headers: { "content-type": "text/html; charset=utf-8" },
      });
    }
  },
};

/** Utilities for Russian phone numbers, safe for both client and server. */

/** Keeps only digits and normalises to the 11-digit 7XXXXXXXXXX form. */
export function normalizePhone(input: string): string | null {
  let digits = input.replace(/\D/g, "");
  if (digits.length === 11 && digits.startsWith("8")) digits = "7" + digits.slice(1);
  if (digits.length === 10) digits = "7" + digits;
  if (digits.length !== 11 || !digits.startsWith("7")) return null;
  return digits;
}

/** Formats digits (without the leading 7) as +7 (999) 123-45-67 */
export function formatPhoneInput(digits: string): string {
  const d = digits.replace(/\D/g, "").slice(0, 10);
  let out = "+7";
  if (d.length > 0) out += ` (${d.slice(0, 3)}`;
  if (d.length >= 3) out += ")";
  if (d.length > 3) out += ` ${d.slice(3, 6)}`;
  if (d.length > 6) out += `-${d.slice(6, 8)}`;
  if (d.length > 8) out += `-${d.slice(8, 10)}`;
  return out;
}

export function prettyPhone(phone: string): string {
  return formatPhoneInput(phone.slice(1));
}

import { createServerFn } from "@tanstack/react-start";

import { normalizePhone } from "./phone";

/**
 * Test mode: the SMS code is returned to the browser and shown on screen.
 * Swap this for a real SMS provider (SMS.ru etc.) when credentials exist.
 */
const TEST_MODE = true;
const CODE_TTL_MS = 5 * 60 * 1000;
const RESEND_COOLDOWN_MS = 60 * 1000;
const MAX_ATTEMPTS = 5;

async function sha256Hex(value: string): Promise<string> {
  const bytes = new TextEncoder().encode(value);
  const digest = await crypto.subtle.digest("SHA-256", bytes);
  return Array.from(new Uint8Array(digest))
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("");
}

function serverSecret(): string {
  return process.env["SUPABASE_SERVICE_ROLE_KEY"] ?? "fallback-dev-secret";
}

function hashCode(phone: string, code: string) {
  return sha256Hex(`${phone}:${code}:${serverSecret()}`);
}

/** Deterministic per-phone password; never leaves the server. */
async function phonePassword(phone: string) {
  const hex = await sha256Hex(`pwd:${phone}:${serverSecret()}`);
  return `Ph!${hex.slice(0, 40)}`;
}

function syntheticEmail(phone: string) {
  return `${phone}@phone.local`;
}

export const requestOtp = createServerFn({ method: "POST" })
  .inputValidator((data: { phone: string }) => data)
  .handler(async ({ data }) => {
    const phone = normalizePhone(data.phone);
    if (!phone) throw new Error("Некорректный номер телефона");

    const { supabaseAdmin } = await import("@/integrations/supabase/client.server");

    const { data: recent } = await supabaseAdmin
      .from("phone_otp")
      .select("created_at")
      .eq("phone", phone)
      .order("created_at", { ascending: false })
      .limit(1)
      .maybeSingle();

    if (recent) {
      const elapsed = Date.now() - new Date(recent.created_at).getTime();
      if (elapsed < RESEND_COOLDOWN_MS) {
        throw new Error(
          `Код уже отправлен. Повторите через ${Math.ceil((RESEND_COOLDOWN_MS - elapsed) / 1000)} сек.`,
        );
      }
    }

    const code = String(Math.floor(100000 + Math.random() * 900000));
    const { error } = await supabaseAdmin.from("phone_otp").insert({
      phone,
      code_hash: await hashCode(phone, code),
      expires_at: new Date(Date.now() + CODE_TTL_MS).toISOString(),
    });
    if (error) throw new Error("Не удалось отправить код, попробуйте ещё раз");

    return { phone, testCode: TEST_MODE ? code : null };
  });

export const verifyOtp = createServerFn({ method: "POST" })
  .inputValidator((data: { phone: string; code: string }) => data)
  .handler(async ({ data }) => {
    const phone = normalizePhone(data.phone);
    const code = data.code.replace(/\D/g, "");
    if (!phone) throw new Error("Некорректный номер телефона");
    if (code.length !== 6) throw new Error("Введите 6 цифр из SMS");

    const { supabaseAdmin } = await import("@/integrations/supabase/client.server");

    const { data: otp } = await supabaseAdmin
      .from("phone_otp")
      .select("id, code_hash, expires_at, attempts, consumed_at")
      .eq("phone", phone)
      .order("created_at", { ascending: false })
      .limit(1)
      .maybeSingle();

    if (!otp || otp.consumed_at) throw new Error("Запросите код заново");
    if (new Date(otp.expires_at).getTime() < Date.now()) throw new Error("Код истёк, запросите новый");
    if (otp.attempts >= MAX_ATTEMPTS) throw new Error("Слишком много попыток, запросите новый код");

    if ((await hashCode(phone, code)) !== otp.code_hash) {
      await supabaseAdmin
        .from("phone_otp")
        .update({ attempts: otp.attempts + 1 })
        .eq("id", otp.id);
      throw new Error("Неверный код");
    }

    await supabaseAdmin
      .from("phone_otp")
      .update({ consumed_at: new Date().toISOString() })
      .eq("id", otp.id);

    const email = syntheticEmail(phone);
    const password = await phonePassword(phone);

    const { createClient } = await import("@supabase/supabase-js");
    const url = process.env["SUPABASE_URL"]!;
    const publishable = process.env["SUPABASE_PUBLISHABLE_KEY"]!;
    const anonClient = createClient(url, publishable, {
      auth: { persistSession: false, autoRefreshToken: false },
      global: {
        fetch: (input: RequestInfo | URL, init?: RequestInit) => {
          const headers = new Headers(init?.headers);
          if (publishable.startsWith("sb_") && headers.get("Authorization") === `Bearer ${publishable}`) {
            headers.delete("Authorization");
          }
          headers.set("apikey", publishable);
          return fetch(input, { ...init, headers });
        },
      },
    });

    let signIn = await anonClient.auth.signInWithPassword({ email, password });

    if (signIn.error) {
      const created = await supabaseAdmin.auth.admin.createUser({
        email,
        password,
        email_confirm: true,
        user_metadata: { phone },
      });
      if (created.error && !created.error.message.toLowerCase().includes("already")) {
        throw new Error("Не удалось создать аккаунт");
      }
      signIn = await anonClient.auth.signInWithPassword({ email, password });
      if (signIn.error || !signIn.data.session) throw new Error("Не удалось войти, попробуйте ещё раз");
    }

    const session = signIn.data.session!;
    const userId = session.user.id;

    await supabaseAdmin
      .from("profiles")
      .upsert({ id: userId, phone }, { onConflict: "id" });

    const { data: profile } = await supabaseAdmin
      .from("profiles")
      .select("full_name, active_mode")
      .eq("id", userId)
      .maybeSingle();

    const { data: roles } = await supabaseAdmin
      .from("user_roles")
      .select("role")
      .eq("user_id", userId);

    return {
      access_token: session.access_token,
      refresh_token: session.refresh_token,
      hasName: Boolean(profile?.full_name),
      roles: (roles ?? []).map((r) => r.role as "client" | "master"),
      activeMode: (profile?.active_mode ?? "client") as "client" | "master",
    };
  });

import { createServerFn } from "@tanstack/react-start";

import { requireSupabaseAuth } from "@/integrations/supabase/auth-middleware";

export type AccountMode = "client" | "master";

export const getMe = createServerFn({ method: "GET" })
  .middleware([requireSupabaseAuth])
  .handler(async ({ context }) => {
    const { supabase, userId } = context;

    const { data: profile } = await supabase
      .from("profiles")
      .select("id, full_name, phone, avatar_url, city, active_mode")
      .eq("id", userId)
      .maybeSingle();

    const { data: roles } = await supabase.from("user_roles").select("role").eq("user_id", userId);

    return {
      profile,
      roles: (roles ?? []).map((r) => r.role as AccountMode),
      activeMode: (profile?.active_mode ?? "client") as AccountMode,
    };
  });

/** Adds a role to the account (idempotent) and makes it the active mode. */
export const switchMode = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((data: { mode: AccountMode }) => data)
  .handler(async ({ data, context }) => {
    const { supabase, userId } = context;
    if (data.mode !== "client" && data.mode !== "master") throw new Error("Неизвестный режим");

    const { data: existing } = await supabase
      .from("user_roles")
      .select("id")
      .eq("user_id", userId)
      .eq("role", data.mode)
      .maybeSingle();

    if (!existing) {
      const { error } = await supabase.from("user_roles").insert({ user_id: userId, role: data.mode });
      if (error) throw new Error("Не удалось сохранить роль");
    }

    const { error: modeError } = await supabase
      .from("profiles")
      .update({ active_mode: data.mode })
      .eq("id", userId);
    if (modeError) throw new Error("Не удалось переключить режим");

    return { mode: data.mode };
  });

export const saveName = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((data: { name: string }) => data)
  .handler(async ({ data, context }) => {
    const name = data.name.trim().slice(0, 60);
    if (name.length < 2) throw new Error("Введите имя");

    const { error } = await context.supabase
      .from("profiles")
      .update({ full_name: name })
      .eq("id", context.userId);
    if (error) throw new Error("Не удалось сохранить имя");

    return { name };
  });

import { createMiddleware } from "@tanstack/react-start";

import { supabase } from "@/integrations/supabase/client";

/** Attaches the signed-in user's bearer token to every server function call. */
export const attachSupabaseAuth = createMiddleware({ type: "function" }).client(async ({ next }) => {
  const { data } = await supabase.auth.getSession();
  const token = data.session?.access_token;
  return next(token ? { headers: { Authorization: `Bearer ${token}` } } : {});
});

import { useNavigate } from "@tanstack/react-router";
import { useServerFn } from "@tanstack/react-start";
import { useState } from "react";

import { supabase } from "@/integrations/supabase/client";
import { switchMode, type AccountMode } from "@/lib/profile.functions";

export function ModeSwitcher({ current }: { current: AccountMode }) {
  const navigate = useNavigate();
  const setMode = useServerFn(switchMode);
  const [loading, setLoading] = useState(false);
  const target: AccountMode = current === "client" ? "master" : "client";

  async function toggle() {
    setLoading(true);
    try {
      await setMode({ data: { mode: target } });
      navigate({ to: target === "master" ? "/master" : "/client" });
    } finally {
      setLoading(false);
    }
  }

  async function signOut() {
    await supabase.auth.signOut();
    navigate({ to: "/", replace: true });
  }

  return (
    <div className="mt-8 space-y-3">
      <button
        onClick={toggle}
        disabled={loading}
        className="w-full rounded-2xl border border-border bg-card px-6 py-4 text-base font-medium text-foreground transition disabled:opacity-50"
      >
        {target === "master" ? "Переключиться в режим мастера" : "Переключиться в режим клиента"}
      </button>
      <button onClick={signOut} className="w-full py-2 text-sm text-muted-foreground">
        Выйти
      </button>
    </div>
  );
}

import { createFileRoute, useNavigate } from "@tanstack/react-router";
import { useServerFn } from "@tanstack/react-start";
import { useEffect, useRef, useState } from "react";

import { supabase } from "@/integrations/supabase/client";
import { requestOtp, verifyOtp } from "@/lib/auth.functions";
import { prettyPhone } from "@/lib/phone";

export const Route = createFileRoute("/auth/code")({
  validateSearch: (search: Record<string, unknown>) => ({
    phone: String(search["phone"] ?? ""),
    test: String(search["test"] ?? ""),
  }),
  head: () => ({
    meta: [
      { title: "Код из SMS — вход" },
      { name: "description", content: "Введите код подтверждения для входа в приложение записи к мастерам." },
      { property: "og:title", content: "Код из SMS — вход" },
      { property: "og:description", content: "Подтверждение номера телефона." },
    ],
  }),
  component: CodeScreen,
});

function CodeScreen() {
  const { phone, test } = Route.useSearch();
  const navigate = useNavigate();
  const verify = useServerFn(verifyOtp);
  const resend = useServerFn(requestOtp);

  const [digits, setDigits] = useState<string[]>(Array(6).fill(""));
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [seconds, setSeconds] = useState(60);
  const [testCode, setTestCode] = useState(test);
  const inputs = useRef<(HTMLInputElement | null)[]>([]);

  useEffect(() => {
    if (seconds <= 0) return;
    const t = setTimeout(() => setSeconds((s) => s - 1), 1000);
    return () => clearTimeout(t);
  }, [seconds]);

  useEffect(() => {
    if (!phone) navigate({ to: "/" });
  }, [phone, navigate]);

  async function submit(code: string) {
    setLoading(true);
    setError(null);
    try {
      const res = await verify({ data: { phone, code } });
      const { error: sessionError } = await supabase.auth.setSession({
        access_token: res.access_token,
        refresh_token: res.refresh_token,
      });
      if (sessionError) throw new Error("Не удалось войти, попробуйте ещё раз");

      if (res.roles.length === 0) {
        navigate({ to: "/auth/role" });
      } else if (!res.hasName) {
        navigate({ to: "/auth/name" });
      } else {
        navigate({ to: res.activeMode === "master" ? "/master" : "/client" });
      }
    } catch (err) {
      setError(err instanceof Error ? err.message : "Что-то пошло не так");
      setDigits(Array(6).fill(""));
      inputs.current[0]?.focus();
    } finally {
      setLoading(false);
    }
  }

  function onChange(index: number, value: string) {
    const chars = value.replace(/\D/g, "");
    if (!chars) {
      const next = [...digits];
      next[index] = "";
      setDigits(next);
      return;
    }
    const next = [...digits];
    for (let i = 0; i < chars.length && index + i < 6; i++) next[index + i] = chars[i]!;
    setDigits(next);
    const filled = Math.min(index + chars.length, 5);
    inputs.current[filled]?.focus();
    const code = next.join("");
    if (code.length === 6 && !next.includes("")) void submit(code);
  }

  async function onResend() {
    setError(null);
    try {
      const res = await resend({ data: { phone } });
      setTestCode(res.testCode ?? "");
      setSeconds(60);
      setDigits(Array(6).fill(""));
    } catch (err) {
      setError(err instanceof Error ? err.message : "Не удалось отправить код");
    }
  }

  return (
    <main className="flex min-h-screen flex-col bg-background px-6 pb-10 pt-16">
      <h1 className="text-3xl font-semibold text-foreground">Код из SMS</h1>
      <p className="mt-3 text-base text-muted-foreground">
        Отправили на номер {phone ? prettyPhone(phone) : ""}
      </p>

      {testCode ? (
        <div className="mt-6 rounded-2xl border border-dashed border-primary/50 bg-accent/50 p-4 text-center">
          <p className="text-xs uppercase tracking-wide text-muted-foreground">Тестовый режим</p>
          <p className="mt-1 text-3xl font-semibold tracking-[0.3em] text-foreground">{testCode}</p>
        </div>
      ) : null}

      <div className="mt-8 flex justify-between gap-2">
        {digits.map((d, i) => (
          <input
            key={i}
            ref={(el) => {
              inputs.current[i] = el;
            }}
            value={d}
            inputMode="numeric"
            maxLength={6}
            onChange={(e) => onChange(i, e.target.value)}
            onKeyDown={(e) => {
              if (e.key === "Backspace" && !digits[i] && i > 0) inputs.current[i - 1]?.focus();
            }}
            className="h-16 w-full rounded-2xl border border-input bg-card text-center text-2xl font-semibold text-foreground outline-none transition focus:border-primary focus:ring-2 focus:ring-ring/30"
          />
        ))}
      </div>

      {error ? <p className="mt-4 text-sm text-destructive">{error}</p> : null}
      {loading ? <p className="mt-4 text-sm text-muted-foreground">Проверяем код…</p> : null}

      <button
        onClick={onResend}
        disabled={seconds > 0}
        className="mt-8 text-sm text-primary disabled:text-muted-foreground"
      >
        {seconds > 0 ? `Отправить снова через ${seconds} сек` : "Отправить код ещё раз"}
      </button>

      <button onClick={() => navigate({ to: "/" })} className="mt-4 text-sm text-muted-foreground">
        Изменить номер
      </button>
    </main>
  );
}
